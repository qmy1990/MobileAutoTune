cmake_minimum_required(VERSION 3.22)
project(MobileAudioAndroid LANGUAGES C CXX)
get_filename_component(MAT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(MAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${MAT_ROOT} ${CMAKE_BINARY_DIR}/core)
set(OBOE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${MAT_ROOT}/third_party/oboe ${CMAKE_BINARY_DIR}/oboe)
add_library(mobileaudio_jni SHARED jni.cpp)
target_include_directories(mobileaudio_jni PRIVATE ${MAT_ROOT}/src)
target_link_libraries(mobileaudio_jni PRIVATE mobileautotune oboe android log)
target_compile_options(mobileaudio_jni PRIVATE -O3 -Wall -Wextra)
target_compile_definitions(mobileaudio_jni PRIVATE $<$<CONFIG:Debug>:MAT_ANDROID_DIAGNOSTICS>)
